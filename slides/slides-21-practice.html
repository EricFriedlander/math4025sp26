Setup

Same voter data setup as notebooks 18–20.

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold, RandomizedSearchCV
from sklearn.preprocessing import OrdinalEncoder, OneHotEncoder
from sklearn.impute import SimpleImputer
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.metrics import accuracy_score, ConfusionMatrixDisplay
from scipy.stats import randint, loguniform
import warnings

warnings.filterwarnings('ignore')
np.random.seed(4025)

# Load and clean
voter_url = 'https://raw.githubusercontent.com/fivethirtyeight/data/master/non-voters/nonvoters_data.csv'
voter_raw = pd.read_csv(voter_url)

voter_clean = voter_raw.drop(columns=['RespId', 'weight', 'Q1']).copy()
voter_clean['educ'] = pd.Categorical(voter_clean['educ'],
    categories=['High school or less', 'Some college', 'College'], ordered=True)
voter_clean['income_cat'] = pd.Categorical(voter_clean['income_cat'],
    categories=['Less than $40k', '$40-75k ', '$75-125k', '$125k or more'], ordered=True)
voter_clean['voter_category'] = pd.Categorical(voter_clean['voter_category'],
    categories=['rarely/never', 'sporadic', 'always'], ordered=False)

voter_clean = voter_clean[voter_clean['Q22'] != 5].copy()
voter_clean['Q22'] = voter_clean['Q22'].fillna('Not Asked').astype(str)

for c in [f'Q28_{i}' for i in range(1, 9)]:
    voter_clean[c] = voter_clean[c].replace(-1.0, 0).fillna('Not Asked').astype(str)
for c in [f'Q29_{i}' for i in range(1, 11)]:
    voter_clean[c] = voter_clean[c].replace(-1.0, 0).fillna('Not Asked').astype(str)

def _party_id(row):
    q31, q32, q33 = row['Q31'], row['Q32'], row['Q33']
    if   q31 == 1: return 'Strong Republican'
    elif q31 == 2: return 'Republican'
    elif q32 == 1: return 'Strong Democrat'
    elif q32 == 2: return 'Democrat'
    elif q33 == 1: return 'Lean Republican'
    elif q33 == 2: return 'Lean Democrat'
    else:          return 'Other'

voter_clean['Party_ID'] = voter_clean.apply(_party_id, axis=1)
party_order = ['Strong Republican', 'Republican', 'Lean Republican',
               'Other', 'Lean Democrat', 'Democrat', 'Strong Democrat']
voter_clean['Party_ID'] = pd.Categorical(voter_clean['Party_ID'],
                                         categories=party_order, ordered=True)
voter_clean = voter_clean.drop(columns=['Q31', 'Q32', 'Q33'])

for c in voter_clean.columns:
    if c != 'ppage' and voter_clean[c].dtype in ['float64', 'int64']:
        voter_clean[c] = voter_clean[c].replace(-1, np.nan)

X_voter = voter_clean.drop(columns=['voter_category'])
y_voter = voter_clean['voter_category']

X_train, X_test, y_train, y_test = train_test_split(
    X_voter, y_voter, test_size=0.3, stratify=y_voter, random_state=4025)

ordered_cat_cols = ['educ', 'income_cat', 'Party_ID']
num_cols = X_train.select_dtypes(include=['int64', 'float64']).columns.tolist()
nom_cols = [c for c in X_train.columns
            if c not in ordered_cat_cols and c not in num_cols]

preproc_voter = ColumnTransformer([
    ('num', Pipeline([('impute', SimpleImputer(strategy='median'))]), num_cols),
    ('ord', Pipeline([
        ('impute', SimpleImputer(strategy='most_frequent')),
        ('encode', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1))
    ]), ordered_cat_cols),
    ('nom', Pipeline([
        ('impute', SimpleImputer(strategy='most_frequent')),
        ('encode', OneHotEncoder(drop='first', sparse_output=False, handle_unknown='ignore'))
    ]), nom_cols),
], remainder='drop')

print(f"Training: {X_train.shape[0]} obs  |  Test: {X_test.shape[0]} obs")

Define Pipeline

Wrap HistGradientBoostingClassifier in the same pipeline pattern used for bagging and random forests.

boost_pipe = Pipeline([
    ('preproc', preproc_voter),
    ('boost', HistGradientBoostingClassifier(random_state=4025))
])
boost_pipe

Random Search

Define a random search over four hyperparameters: - max_iter: total number of trees (50–400) - learning_rate: log-uniform between 0.001 and 0.5 (small values = more trees needed) - max_depth: depth of each individual tree (2–10) - min_samples_leaf: minimum observations required in a leaf (10–100)

We sample 50 random combinations and evaluate each with 5-fold stratified cross-validation.

param_dist = {
    'boost__max_iter':         randint(50, 400),
    'boost__learning_rate':    loguniform(1e-3, 0.5),
    'boost__max_depth':        randint(2, 10),
    'boost__min_samples_leaf': randint(10, 100),
}

boost_cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=4025)

boost_search = RandomizedSearchCV(
    estimator=boost_pipe,
    param_distributions=param_dist,
    n_iter=50,
    cv=boost_cv,
    scoring='accuracy',
    return_train_score=True,
    random_state=4025,
    n_jobs=-1
)
boost_search

Fit

Fit the random search. This evaluates 50 × 5 = 250 model fits. Expect ~30–60 seconds with n_jobs=-1.

boost_search.fit(X_train, y_train)
print("Search complete.")

CV Results

Scatter plot of CV accuracy vs. learning rate (log scale), colored by max_depth. Look for patterns: which learning rate range performs best? Does tree depth interact with learning rate?

boost_results = pd.DataFrame(boost_search.cv_results_).sort_values(
    'mean_test_score', ascending=False
)

fig, ax = plt.subplots(figsize=(7, 4))
sc = ax.scatter(
    boost_results['param_boost__learning_rate'],
    boost_results['mean_test_score'],
    c=boost_results['param_boost__max_depth'],
    cmap='viridis', alpha=0.7
)
ax.set_xscale('log')
ax.set_xlabel('learning_rate (log scale)')
ax.set_ylabel('CV Accuracy')
ax.set_title('Boosting Random Search Results\n(color = max_depth)')
cb = plt.colorbar(sc, ax=ax)
cb.set_label('max_depth')
plt.tight_layout()
plt.show()

Best Parameters

Print the hyperparameter values that achieved the highest CV accuracy.

print("Best hyperparameters:")
for k, v in boost_search.best_params_.items():
    short_k = k.replace('boost__', '')
    print(f"  {short_k}: {v:.4g}" if isinstance(v, float) else f"  {short_k}: {v}")
print(f"\nBest CV Accuracy: {boost_search.best_score_:.3f}")

Test Set Evaluation

Evaluate the best model on the held-out test set. Compare accuracy and confusion matrix to the random forest from notebook 19.

boost_preds = boost_search.predict(X_test)
print(f"Test Accuracy: {accuracy_score(y_test, boost_preds):.3f}")

fig, ax = plt.subplots(figsize=(6, 5))
ConfusionMatrixDisplay.from_predictions(
    y_test, boost_preds,
    display_labels=list(y_voter.cat.categories),
    ax=ax, colorbar=False
)
ax.set_title('Confusion Matrix: Gradient Boosting')
plt.tight_layout()
plt.show()

Variable Importance

HistGradientBoostingClassifier provides feature_importances_ (MDI-based) after fitting. Compare the top 20 features to the random forest results from notebook 20.

best_boost = boost_search.best_estimator_.named_steps['boost']
enc = boost_search.best_estimator_.named_steps['preproc'].named_transformers_['nom'].named_steps['encode']
boost_feature_names = num_cols + ordered_cat_cols + list(enc.get_feature_names_out(nom_cols))

boost_imp_df = (
    pd.DataFrame({'feature': boost_feature_names,
                  'importance': best_boost.feature_importances_})
    .sort_values('importance', ascending=False)
    .head(20)
)

fig, ax = plt.subplots(figsize=(7, 6))
ax.barh(boost_imp_df['feature'][::-1], boost_imp_df['importance'][::-1], color='darkorange')
ax.set_xlabel('Feature Importance (MDI)')
ax.set_title('Top 20 Features — Gradient Boosting')
plt.tight_layout()
plt.show()
